# mega26: Spain-France's monarch is the Sun Lord. Loaded first so it wins over the generic monarchies;
# only applies while SPA is a (non-regency) monarchy, otherwise the normal governments take over.
gov_mega26_sun_kingdom = {
	transfer_of_power = hereditary

	male_ruler = "RULER_TITLE_MEGA26_SUN_LORD"
	female_ruler = "RULER_TITLE_MEGA26_SUN_LADY"

	male_heir = "RULER_TITLE_PRINCE"
	female_heir = "RULER_TITLE_PRINCESS"

	possible = {
		c:SPA ?= this
		has_law = law_type:law_monarchy
		NOT = { is_subject_type = subject_type_crown_land }
		has_gov_regency = no
		is_domain_alliance_gov = no
	}

	on_government_type_change = {
		change_to_hereditary = yes
	}
	on_post_government_type_change = {
		post_change_to_hereditary = yes
	}
}
