# mega26: the Empire of China (Altai's formed China) can proclaim the Great Mongolian State.
# A decision with change_tag, so it doesn't depend on the engine letting an already-formed nation use the normal formation.
mega26_proclaim_great_mongolian_state = {
	is_shown = {
		exists = c:CHI
		c:CHI = ROOT
		is_player = yes # mega26: players only - the AI never sees or takes this
		NOT = { exists = c:MGL }
	}

	possible = {
		is_subject = no
		is_at_war = no
		any_scope_state = { # same territory as the Mongol formable: 9 of the 14 Mongol homeland states (60%)
			count >= 9
			OR = {
				state_region = s:STATE_URGA
				state_region = s:STATE_ULIASTAI
				state_region = s:STATE_ALXA
				state_region = s:STATE_HINGGAN
				state_region = s:STATE_BEIJING
				state_region = s:STATE_GANSU
				state_region = s:STATE_DZUNGARIA
				state_region = s:STATE_NORTHERN_MANCHURIA
				state_region = s:STATE_SOUTHERN_MANCHURIA
				state_region = s:STATE_QINGHAI
				state_region = s:STATE_JETISY
				state_region = s:STATE_TIANSHAN
				state_region = s:STATE_TRANS_BAIKAL
				state_region = s:STATE_IRKUTSK
			}
		}
	}

	when_taken = {
		change_tag = MGL
		set_state_religion = rel:confucian
	}

	ai_chance = {
		value = 0
	}
}
